### 4. ヘルパー関数（全シナリオ共通）

P1のブラウザコンソールに以下を貼り付ける。

```javascript
// カード生成ヘルパー
function c(id, color, value, points) {
  return { id, color, value, points };
}

// デッキ構築（3人用）
// P1=オーナー(1番目に参加), P2=2番目に参加, P3=3番目に参加
function buildDeck(firstCard, p1Hand, p2Hand, p3Hand) {
  const padding = [];
  const colors = ["red", "blue", "green", "yellow"];
  for (let i = 0; i < 50; i++) {
    const color = colors[i % 4];
    const value = String(i % 10);
    padding.push(c("pad-" + i, color, value, i % 10));
  }
  // deck.pop() で末尾から取り出されるため、末尾にP1の1枚目が来るように配置
  // [padding..., firstCard, P3[6], P2[6], P1[6], ..., P3[0], P2[0], P1[0]]
  const deck = [...padding, firstCard];
  for (let r = 6; r >= 0; r--) {
    deck.push(p3Hand[r]);
    deck.push(p2Hand[r]);
    deck.push(p1Hand[r]);
  }
  return deck;
}

// デッキをセットしてログ出力
function setDeck(firstCard, p1Hand, p2Hand, p3Hand) {
  const deck = buildDeck(firstCard, p1Hand, p2Hand, p3Hand);
  window.__debugRoom.send("__setDeck", deck);
  console.log("デッキセット完了");
  console.log("場札:", firstCard.id, "(points:", firstCard.points + ")");
  console.log("P1手札:", p1Hand.map(c => c.id).join(", "), "| 合計:", p1Hand.reduce((s,c) => s+c.points, 0));
  console.log("P2手札:", p2Hand.map(c => c.id).join(", "), "| 合計:", p2Hand.reduce((s,c) => s+c.points, 0));
  console.log("P3手札:", p3Hand.map(c => c.id).join(", "), "| 合計:", p3Hand.reduce((s,c) => s+c.points, 0));
}
```
